cmake_minimum_required(VERSION 3.20.6...3.31)
# Native Windows custom application. Adapted from SlicerCustomAppTemplate
# 1863e64fe2c622a4de8021b145de180a91d8de52 (Apache-2.0).
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(EP_GIT_PROTOCOL https)
include(FetchContent)
set(MENISCUS_SLICER_REVISION 4e21c19d8360242d88d6ca2490bf52400441b8e1)
if(NOT DEFINED slicersources_SOURCE_DIR)
  FetchContent_Populate(slicersources
    GIT_REPOSITORY https://github.com/Slicer/Slicer.git
    GIT_TAG ${MENISCUS_SLICER_REVISION}
    GIT_PROGRESS 1)
else()
  execute_process(COMMAND git -C "${slicersources_SOURCE_DIR}" rev-parse HEAD
    OUTPUT_VARIABLE actual_revision OUTPUT_STRIP_TRAILING_WHITESPACE
    COMMAND_ERROR_IS_FATAL ANY)
  if(NOT actual_revision STREQUAL MENISCUS_SLICER_REVISION)
    message(FATAL_ERROR "Slicer sources must be pinned to ${MENISCUS_SLICER_REVISION}")
  endif()
  set(slicersources_BINARY_DIR ${CMAKE_CURRENT_BINARY_DIR}/slicersources-subbuild)
endif()
set(CMAKE_MODULE_PATH ${slicersources_SOURCE_DIR}/CMake ${CMAKE_MODULE_PATH})
include(SlicerInitializeOSXVariables)
project(MeniscusAnnotator)
if(NOT WIN32)
  message(FATAL_ERROR "This custom application package targets Windows x64. Use installed Slicer for development on other platforms.")
endif()
set(Slicer_APPLICATIONS_DIR ${CMAKE_CURRENT_SOURCE_DIR}/packaging/Applications)
set(Slicer_MAIN_PROJECT MeniscusAnnotatorApp)
set(Slicer_ORGANIZATION_NAME "Meniscus Research")
set(Slicer_ORGANIZATION_DOMAIN "local.meniscus")
set(Slicer_DEFAULT_HOME_MODULE MeniscusAnnotator)
set(Slicer_DEFAULT_FAVORITE_MODULES MeniscusAnnotator)
set(SUPERBUILD_TOPLEVEL_PROJECT Slicer)
set(EXTERNAL_PROJECT_ADDITIONAL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/packaging/SuperBuild")
include(ExternalProjectDependency)
set(Slicer_ADDITIONAL_DEPENDENCIES python-meniscus-requirements)
set(Slicer_ITK_ADDITIONAL_MODULES "")
set(Slicer_DEFAULT_BUILD_TYPE Release)
include(SlicerInitializeBuildType)
set(Slicer_DEFAULT_RELEASE_TYPE Stable)
include(SlicerInitializeReleaseType)
set(Slicer_CPACK_NSIS_INSTALL_REQUIRES_ADMIN_ACCOUNT OFF)
set(Slicer_CPACK_NSIS_INSTALL_ROOT "$LOCALAPPDATA/Meniscus Research")
option(BUILD_TESTING "Build native test support" ON)
option(Slicer_BUILD_APPLICATIONUPDATE_SUPPORT "Disable runtime updates for fixed annotation batches" OFF)
option(Slicer_BUILD_DOCUMENTATION "Build API documentation" OFF)
option(Slicer_BUILD_EXTENSIONMANAGER_SUPPORT "No runtime extension downloads" OFF)
option(Slicer_BUILD_DICOM_SUPPORT "NIfTI-only input contract" OFF)
option(Slicer_BUILD_DIFFUSION_SUPPORT "Diffusion" OFF)
option(Slicer_BUILD_MULTIVOLUME_SUPPORT "MultiVolume" OFF)
option(Slicer_BUILD_PARAMETERSERIALIZER_SUPPORT "Parameter serializer" OFF)
option(Slicer_USE_PYTHONQT "Python scripting" ON)
option(Slicer_USE_QtTesting "Qt Testing" OFF)
option(Slicer_USE_SimpleITK "Independent saved-file reader" ON)
option(Slicer_BUILD_WIN32_CONSOLE_LAUNCHER "Launcher console" OFF)
option(Slicer_BUILD_WIN32_CONSOLE "Runtime console" OFF)
option(Slicer_BUILD_BRAINSTOOLS "BRAINSTools" OFF)
option(Slicer_BUILD_DataStore "DataStore" OFF)
option(Slicer_BUILD_CompareVolumes "CompareVolumes" OFF)
option(Slicer_BUILD_LandmarkRegistration "LandmarkRegistration" OFF)
option(Slicer_BUILD_SurfaceToolbox "SurfaceToolbox" OFF)
set(Slicer_CLIMODULES_ENABLED "")
set(Slicer_QTLOADABLEMODULES_ENABLED "")
set(Slicer_QTSCRIPTEDMODULES_ENABLED "")
set(Slicer_QTLOADABLEMODULES_DISABLED SceneViews SlicerWelcome)
set(Slicer_QTSCRIPTEDMODULES_DISABLED
  DataProbe DMRIInstall Endoscopy LabelStatistics PerformanceTests SampleData VectorToScalarVolume)
set(Slicer_EXTENSION_SOURCE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/packaging/module")
add_subdirectory(${slicersources_SOURCE_DIR} ${slicersources_BINARY_DIR})
